# 多任务系统

**RTOS可以做到单核cpu让多个任务“并行”跑起来**

为了实现这个功能，RTOS使用链表判断

## 链表-list

RTOS的链表是环状双向链表

```c
//list.c  list.h
|     List_t     | 
||MiniListItem_t||<-->| ListItem_t |<- ... ->| ListItem_t |
         ^                                          ^
         |__________________________________________|
```

RTOS的链表是一个很重要的数据结构，可以认为RTOS就是通过链表进行任务调度的。

ListItem_t中除去指向前一个、后一个元素的指针这些基本成员外，还包含2个对RTOS很重要的成员：

1. 表明当前这个item是挂在那个链表里的
2. 表明这个item是属于哪个任务的（可以通过item反向找到任务）

List_t对象中存在一个只用于链接头尾的节点--MiniListItem，这个特殊的节点只包含了



## 裸机系统和多任务系统

了解多任务系统前，先了解一下单片机裸机执行多个“任务”的方法

```c
// main.c
int main(void)
{
    while(1)
    {
        if(按键1被按下)
        {
            执行任务1
        }
        if(按键2被按下)
        {
            执行任务2
        }
        //可能还有按键3、按键4...
    }
}
```

裸机一般是通过轮询检测事件触发并进行执行的，这种方式有不少问题，比如：

1. cpu需要一直工作，即使没有什么东西条件被触发也必须不断执行循环
2. 一个任务执行，整个系统都会被卡住，直到这个任务执行完成

RTOS是多任务系统，相较于裸机系统，RTOS可以通过任务调度，让任务“并行”（当然只是看起来像是在并行...单核CPU无论怎么样都不可能实现真正的并行）



## 任务的组成部分

RTOS中，任务的本质是一个不断循环，且永远不返回的函数

```c
void task_entry (void *prag)
{
    for(;;)//不断循环
    {
        //任务主体
    }
}
```

但是为了能让CPU实现任务调度，还需要一些其他组件配合。

### 要素1：任务栈

任务的调度意味着在执行任务1的时候可能被打断，CPU跑去执行任务2，这就需要一个地方保存任务1的现场参数，否则任务1的现场酒永远都没办法恢复了

RTOS把任务的现场被保存在任务栈中，任务栈是需要定义的

```c
// 定义任务栈
#define TASK1_STACK_SIZE                    128      
StackType_t Task1Stack[TASK1_STACK_SIZE];            
  
#define TASK2_STACK_SIZE                    128 
StackType_t Task2Stack[TASK2_STACK_SIZE]; 
```

任务栈本质上是一个数组，用于在切换的时候给被打断的任务做“存档”

### 要素2：任务函数

```c
void delay (uint32_t count)
```

可见任务函数是无限循环且永不返回的



### 要素3：任务控制块TCB

任务控制块--Task Control Block

作用：保存任务信息（任务栈指针、任务名称、任务的形参......）

RTOS定义了一个结构体用于描述任务控制块

```c
typedef struct tskTaskControlBlock 
{ 
    volatile StackType_t *pxTopOfStack; /* 栈顶 */
    ListItem_t xStateListItem; /* 任务节点 */
    StackType_t *pxStack; /* 任务栈起始地址 */ 
    char pcTaskName[ configMAX_TASK_NAME_LEN ];  /* 任务名称，字符串形式 */
} tskTCB; 
typedef tskTCB TCB_t;
```

### 创建任务

任务栈

任务实体函数

任务的控制块

需要联合起来才能使用，这个过程就是创建任务的过程

RTOS提供两种创建任务的方式，动态创建和静态创建，动态创建TCB和任务栈是动态分配的，删除任务的时候将释放内存；静态创建则需要预先手动写好，任务删除时也不会释放内存

调用过程：

xTaskCreateStatic -> prvInitialiseNewTask -> pxPortInitialiseStack

​      创建任务         ->         初始化任务        ->     初始化任务堆栈

任务栈、任务、TCB这三个要素，除了任务函数在进行创建任务之前是写好的；任务栈、TCB则是定义了一个变量，在创建任务的过程中初始化他们两个；

```c
#if( configSUPPORT_STATIC_ALLOCATION == 1 ) //静态任务创建

/* 任务栈、任务控制块、任务函数都是现成的，送到 xTaskCreateStatic 里打包*/
TaskHandle_t xTaskCreateStatic( TaskFunction_t pxTaskCode, // 任务函数
                               //（返回类型为void，参数为void*的函数指针）
                               // typedef void (*TaskFunction_t)( void * )
                                const char * const pcName, //任务名         
                                const uint32_t ulStackDepth,//任务栈大	小，单位为word       
                                void * const pvParameters,// 任务形参
                                StackType_t * const puxStackBuffer,//任务栈起始地址
                                TCB_t * const pxTaskBuffer ) //任务控制块指针      
{ 
    TCB_t *pxNewTCB; 
    TaskHandle_t xReturn; // 其实是个void指针 typedef void * TaskHandle_t;
 
    if ( ( pxTaskBuffer != NULL ) && ( puxStackBuffer != NULL ) ) 
    { 
        pxNewTCB = ( TCB_t * ) pxTaskBuffer; 
        pxNewTCB->pxStack = ( StackType_t * ) puxStackBuffer; 
 
        /* 创建新的任务 */
        prvInitialiseNewTask( pxTaskCode,        /* 任务入口 */ 
                              pcName,            /* 任务名称，字符串形式 */ 
                              ulStackDepth,      /* 任务栈大小，单位为字 */ 
                              pvParameters,      /* 任务形参 */ 
                              &xReturn,          /* 任务句柄 */ 
                              pxNewTCB);         /* 任务控制块指针 */ 
    } 
    else 
    { 
        xReturn = NULL; 
    } 

    /* 返回任务句柄，如果任务创建成功，此时xReturn应该指向任务控制块 */ 
    return xReturn;
} 
 
#endif /* configSUPPORT_STATIC_ALLOCATION */                             

```



### 初始化任务

把栈顶地址、任务名称、任务栈塞到任务控制块TCB_t里

```c
static void prvInitialiseNewTask(TaskFunction_t pxTaskCode,
                                 const char * const pcName,
                                 const uint32_t ulStackDepth,
                                 void * const pvParameters,
                                 TaskHandle_t * const pxCreatedTask,
                                 TCB_t *pxNewTCB )

{
    StackType_t *pxTopOfStack;
    UBaseType_t x;
    /* 获取栈顶地址 */
    pxTopOfStack = pxNewTCB->pxStack + ( ulStackDepth - ( uint32_t ) 1 );
    /* 向下做8字节对齐 */
    pxTopOfStack = ( StackType_t * ) \
        ( ( ( uint32_t ) pxTopOfStack ) & ( ~( ( uint32_t ) 0x0007 ) ) );
    /* 将任务的名字存储在TCB中 */
    for ( x = ( UBaseType_t ) 0; x < ( UBaseType_t ) configMAX_TASK_NAME_LEN; x++ )
    {
        pxNewTCB->pcTaskName[ x ] = pcName[ x ];
        if ( pcName[ x ] == 0x00 )
        {
            break;
        }
    }
    /* 任务名字的长度不能超过configMAX_TASK_NAME_LEN */
    pxNewTCB->pcTaskName[ configMAX_TASK_NAME_LEN - 1 ] = '\0';
    /* 初始化TCB中的xStateListItem节点 */
    vListInitialiseItem( &( pxNewTCB->xStateListItem ) );
    /* 设置xStateListItem节点的拥有者 */
    listSET_LIST_ITEM_OWNER( &( pxNewTCB->xStateListItem ), pxNewTCB );
    /* 初始化任务栈 */
    pxNewTCB->pxTopOfStack = pxPortInitialiseStack( pxTopOfStack,
                            pxTaskCode,
                            pvParameters );
    /* 让任务句柄指向任务控制块 */
    if ( ( void * ) pxCreatedTask != NULL )
    {
        *pxCreatedTask = ( TaskHandle_t ) pxNewTCB;
    }
}
```



### 初始化任务堆栈

```c
#define portINITIAL_XPSR            ( 0x01000000 )
#define portSTART_ADDRESS_MASK      ( ( StackType_t ) 0xfffffffeUL )

static void prvTaskExitError( void )
{
    /* 函数停止在这里 */
    for (;;);
}

StackType_t *pxPortInitialiseStack(StackType_t *pxTopOfStack,
                                   TaskFunction_t pxCode,
                                   void *pvParameters )
{
    /* 异常发生时，自动加载到CPU寄存器的内容 */(1)
    pxTopOfStack--;
    *pxTopOfStack = portINITIAL_XPSR;(2)
    pxTopOfStack--;
    *pxTopOfStack = ( ( StackType_t ) pxCode ) & portSTART_ADDRESS_MASK;(3)
    pxTopOfStack--;
    *pxTopOfStack = ( StackType_t ) prvTaskExitError;(4) 
    // 这对应的是LR，但这里填写的prvTaskExitError是一个无限循环的函数
    // 这表明使用这个栈时，没有真正的返回，只有原地打转
    pxTopOfStack -= 5;      /* R12, R3, R2 and R1 默认初始化为0 */
    *pxTopOfStack = ( StackType_t ) pvParameters;   (5)
    /* 异常发生时，手动加载到CPU寄存器的内容 */(6)
    pxTopOfStack -= 8;
    /* 返回栈顶指针，此时pxTopOfStack指向空闲栈 */
    return pxTopOfStack;(7)
}
```

```c
/*
*参考资料《STM32F10xxx Cortex-M3 programming manual》4.4.3，百度搜索“PM0056”即可找到这个文档
* 在Cortex-M中，内核外设SCB中SHPR3寄存器用于设置SysTick和PendSV的异常优先级
* System handler priority register 3 (SCB_SHPR3) SCB_SHPR3：0xE000 ED20
* Bits 31:24 PRI_15[7:0]: Priority of system handler 15, SysTick exception
* Bits 23:16 PRI_14[7:0]: Priority of system handler 14, PendSV
*/
#define portNVIC_SYSPRI2_REG         (*(( volatile uint32_t *) 0xe000ed20))

#define portNVIC_PENDSV_PRI(((uint32_t) configKERNEL_INTERRUPT_PRIORITY ) << 16UL)
#define portNVIC_SYSTICK_PRI(((uint32_t) configKERNEL_INTERRUPT_PRIORITY ) << 24UL )

BaseType_t xPortStartScheduler( void )
{
    /* 配置PendSV 和 SysTick 的中断优先级为最低 */
    portNVIC_SYSPRI2_REG |= portNVIC_PENDSV_PRI;
    portNVIC_SYSPRI2_REG |= portNVIC_SYSTICK_PRI;

    /* 启动第一个任务，不再返回 */
    prvStartFirstTask();(2)

    /* 不应该运行到这里 */
    return 0;
}
```



```c
/*
* 参考资料《STM32F10xxx Cortex-M3 programming manual》4.4.3，百度搜索“PM0056”即可找到这个文档
* 在Cortex-M中，内核外设SCB的地址范围为：0xE000ED00-0xE000ED3F
* 0xE000ED008为SCB外设中SCB_VTOR这个寄存器的地址，里面存放的是向量表的起始地址，即MSP的地址
*/
__asm void prvStartFirstTask( void ) // asm表明用到了汇编
{
    PRESERVE8

/* 在Cortex-M中，0xE000ED08是SCB_VTOR这个寄存器的地址
里面存放的是向量表的起始地址，即MSP的地址 */
    ldr r0, =0xE000ED08 // 这个地址是VTOR寄存器的地址
    ldr r0, [r0] // 取VTOR里保存的向量表的地址
    ldr r0, [r0] // 取向量表中的中断服务的地址

    /* 设置主栈指针msp的值 */
    msr msp, r0 // 将VTOR的寄存器地址给MSP

    /* 使能全局中断 */
    cpsie i
    cpsie f
    dsb
    isb

    /* 调用SVC去启动第一个任务 */
    svc 0 //！！！！！这里很重要！！！！！！！！！
    nop
    nop
}
```

```c
__asm void vPortSVCHandler( void ) // SVC中断服务处理函数
{
    extern pxCurrentTCB;(1)// pxCurrentTCB是定义在外部的一个指示当前TCB的指针
    // 这个pxCurrentTCB也就是最开始我们指定的第一个任务
    PRESERVE8 // 8字节对齐，为了应对浮点运算
    ldr r3, =pxCurrentTCB(2) // 加载pxCurrentTCB这个指针变量本身的地址
    ldr r1, [r3](3) // 加载指针指向的内容，也就是一个TCB的地址
    ldr r0, [r1](4) // TCB第一个元素的地址，也就是pxTopOfStack的地址
    /* 这里必须解释一下pxTopOfStack是手动申请的一个数组的数组头地址 */
    ldmia r0!, {r4-r11}(5) //将内存里的内容写到
    /* 内存 -> 寄存器 */
    /* ldmia将r0指向的内存写入r4后，r0的地址会+4 */
    /* 这种操作方式很适合操作数组 */
    /* 最后！会让r0回到最开始的位置 */
    msr psp, r0(6) // 将栈顶地址给psp
    isb // 清理流水线
    mov r0, #0(7) // r0赋值为0
    msr basep ri, r0(8) // 核心对中断的关断状态，0为不关断任何中断

    // 也就是，之前的程序用的是MSP还是PSP，thread模式还是handle模式
    bx r14(10) // 返回
}
```

### 真正的任务切换函数

任务切换的本质：RTOS根据就绪队列里的优先级，寻找优先级最高的下一个任务，然后进行切换

```c
/* 在task.h中定义 */
#define taskYIELD()                 portYIELD()

/* 在portmacro.h中定义 */
/* 中断控制状态寄存器：0xe000ed04
* Bit 28 PENDSVSET: PendSV 悬起位
*/
#define portNVIC_INT_CTRL_REG               (*(( volatile uint32_t *) 0xe000ed04))
#define portNVIC_PENDSVSET_BIT              ( 1UL << 28UL )

#define portSY_FULL_READ_WRITE              ( 15 )

#define portYIELD()
{
    /* 触发PendSV，产生上下文切换 */
    portNVIC_INT_CTRL_REG = portNVIC_PENDSVSET_BIT;// 将pendsv的标志位挂起
    __dsb( portSY_FULL_READ_WRITE );
    __isb( portSY_FULL_READ_WRITE );
}
```



```c
// 这才是真正用来切换任务的函数
__asm void xPortPendSVHandler( void )
{
    extern pxCurrentTCB;(1)
    extern vTaskSwitchContext;(2)

    PRESERVE8(3)

    mrs r0, psp(4) //将psp的值给r0
    isb //清空流水线

    ldr r3, =pxCurrentTCB(5)
    ldr r2, [r3](6) // 此时R2里的内容就是TCB的真实地址

    stmdb r0!, {r4-r11}(7) // 将r4到r11的值写入r0为基地的地址中（寄存器->内存）
    str r0, [r2](8) // 把当前TopOfStack的地址写入r0

    stmdb sp!, {r3, r14}(9) // 把r3（TCB）和lr写入sp（msp）
    mov r0, #configMAX_SYSCALL_INTERRUPT_PRIORITY(10) // 决定屏蔽某些优先级的中断
    msr basepri, r0(11) // 执行屏蔽
    dsb
    isb
    bl vTaskSwitchContext(12) // 跳转到vTaskSwitchContext函数
        /*这一步之后 pxCurrentTCB就被切换了 */
    mov r0, #0(13) // 取消对中断优先级的屏蔽
    msr basepri, r0// 执行屏蔽取消
    ldmia sp!, {r3, r14}(14) // 把r3和lr压入栈中
    /*这里使用的MSP*/
    /*中断服务里的sp都是MSP*/

    ldr r1, [r3](15) // 取pxCurrentTCB指向的TCB
    ldr r0, [r1](16) // 取该TCB的第一个内容，即TopOfStack
    ldmia r0!, {r4-r11}(17) // 将r4到r11的值写入TopOfStack指示的栈中
    msr psp, r0(18) //重新设定psp到任务2的TopOfStack
    isb
    bx r14(19)
    nop
}
```
