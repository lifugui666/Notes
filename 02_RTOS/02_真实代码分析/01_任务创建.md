## 静态任务创建

### 要点：

1. 静态任务需要手动创建TCB和栈
2. （struct xSTATIC_TCB 即 StaticTask_t） 和 （TCB_t 即 struct tskTaskControlBlock）在内容和形式上是一样的，但StaticTask_t是给用户用的，用户无法通过StaticTask_t看到任务控制块里的各个成员的具体名称（都被隐为xxDummyx的命名形式）；内核真正使用的是TCB_t 
3. 静态任务即使被删除，内存也不允许被释放，否则就是在释放bss段的内存，会炸掉；
4. 代码中大量使用了宏定义的钩子，这些钩子本质上是宏定义，实际项目里可以通过修改这些宏定义，实现输出日志等功能，方便调试；

步骤：

1. 创建TCB
2. 创建栈
3. xTaskCreateStatic() 创建静态任务
   1. prvCreateStaticTask() 创建静态任务
      1. prvInitialiseNewTask() 初始化静态任务
         1. 初始化栈

#### xTaskCreateStatic

**封装：初始化任务栈、初始化TCB；完成初始化之后将任务添加到任务列表中**

```c
TaskHandle_t xTaskCreateStatic( TaskFunction_t pxTaskCode,
                                const char * const pcName,
                                const configSTACK_DEPTH_TYPE uxStackDepth,
                                void * const pvParameters,
                                UBaseType_t uxPriority,
                                StackType_t * const puxStackBuffer,
                                StaticTask_t * const pxTaskBuffer )
{
    TaskHandle_t xReturn = NULL;
    TCB_t * pxNewTCB;

    // 这是一个钩子，本身是一个宏定义，目前它不执行任何任务，如果重构这个宏，可以让它执行输出日志等功能
    traceENTER_xTaskCreateStatic( pxTaskCode, pcName, uxStackDepth, pvParameters, uxPriority, puxStackBuffer, pxTaskBuffer );

    // 初始化栈、TCB
    pxNewTCB = prvCreateStaticTask( pxTaskCode, pcName, uxStackDepth, pvParameters, uxPriority, puxStackBuffer, pxTaskBuffer, &xReturn );

    if( pxNewTCB != NULL )
    {
        #if ( ( configNUMBER_OF_CORES > 1 ) && ( configUSE_CORE_AFFINITY == 1 ) )
        {
            // 设置TCB参数uxCoreAffinityMask，这个参数决定任务能跑在哪些核心上（多核场景）
            pxNewTCB->uxCoreAffinityMask = configTASK_DEFAULT_CORE_AFFINITY;
        }
        #endif
		// 将新任务添加到就绪列表里
        prvAddNewTaskToReadyList( pxNewTCB );
    }
    else
    {
        // 这也是一个钩子，本身是一个宏定义，通过重写这个宏，可以输出一些信息
        mtCOVERAGE_TEST_MARKER();
    }

    // 钩子，本身是一个宏定义，可以用于输出一些信息
    traceRETURN_xTaskCreateStatic( xReturn );

    return xReturn;
}
```

#### prvCreateStaticTask

**封装：创建静态任务，在创建之前检查了一些条件（比如检查TCB内存分布一致性、设置静态任务标志位之类的工作）**

```c
static TCB_t * prvCreateStaticTask( TaskFunction_t pxTaskCode,
                                        const char * const pcName,
                                        const configSTACK_DEPTH_TYPE uxStackDepth,
                                        void * const pvParameters,
                                        UBaseType_t uxPriority,
                                        StackType_t * const puxStackBuffer,
                                        StaticTask_t * const pxTaskBuffer,
                                        TaskHandle_t * const pxCreatedTask )
{
    TCB_t * pxNewTCB;

    // 防御性断言的钩子，目前这两句什么都不做
    // 但是可以通过修改configASSERT宏定义，让这两句做一些你想要的操作
    // 比如写一个无限循环，让代码卡在这里，方便调试
    configASSERT( puxStackBuffer != NULL );
    configASSERT( pxTaskBuffer != NULL );

    // 判断，TCB的size和StaticTask_t的size是否一致，防止升级只修改了其中的一个导致错误
    #if ( configASSERT_DEFINED == 1 )
    {
        volatile size_t xSize = sizeof( StaticTask_t );
        configASSERT( xSize == sizeof( TCB_t ) );
        ( void ) xSize; /* Prevent unused variable warning when configASSERT() is not used. */
    }
    #endif /* configASSERT_DEFINED */

    if( ( pxTaskBuffer != NULL ) && ( puxStackBuffer != NULL ) )
    {
        pxNewTCB = ( TCB_t * ) pxTaskBuffer;
        ( void ) memset( ( void * ) pxNewTCB, 0x00, sizeof( TCB_t ) );
        pxNewTCB->pxStack = ( StackType_t * ) puxStackBuffer;

        #if ( tskSTATIC_AND_DYNAMIC_ALLOCATION_POSSIBLE != 0 )
        {
            //这是在设置TCB的标志位，提醒系统这个TCB是静态的，绝对不允许释放内存
            //否则就是在释放bss段的内存了
            pxNewTCB->ucStaticallyAllocated = tskSTATICALLY_ALLOCATED_STACK_AND_TCB;
        }
        #endif /* tskSTATIC_AND_DYNAMIC_ALLOCATION_POSSIBLE */

        prvInitialiseNewTask( pxTaskCode, pcName, uxStackDepth, pvParameters, uxPriority, pxCreatedTask, pxNewTCB, NULL );
    }
    else
    {
        pxNewTCB = NULL;
    }

    return pxNewTCB;
}
```

#### prvInitialiseNewTask

**这是真正执行任务创建的函数，demo的main函数里只创建了一个静态任务作为例子，实际上这个函数还可以用于创建动态任务**

接下来的代码里出现了MPU相关的设置，MPU是一个硬件单元，如果你使用的SoC有这个硬件并开启了MPU支持，那么就可以：

1. 让程序在 “特权” “用户”阶级之间进行切换
2. 提供硬件层面的栈溢出保护（栈溢出会直接触发异常，不用幸苦软件检测栈溢出）

```c
static void prvInitialiseNewTask( TaskFunction_t pxTaskCode,
                                  const char * const pcName,
                                  const configSTACK_DEPTH_TYPE uxStackDepth,
                                  void * const pvParameters,
                                  UBaseType_t uxPriority,
                                  TaskHandle_t * const pxCreatedTask,
                                  TCB_t * pxNewTCB,
                                  const MemoryRegion_t * const xRegions )
{
    StackType_t * pxTopOfStack;
    UBaseType_t x;

    // 特权模式的标志位，如果portUSING_MPU_WRAPPERS配置为1
    // 那么任务可以被设置为 特权任务
    // 特权任务的优先级参数，最高位为1
    // FreeRTOS的优先级不可能比256大，所以最高位不会影响到真正的优先级数
    // 特权任务是可以进行内核操作的，这需要硬件支持，有些CPU是没有MPU的
    #if ( portUSING_MPU_WRAPPERS == 1 )
        /* Should the task be created in privileged mode? */
        BaseType_t xRunPrivileged;

        if( ( uxPriority & portPRIVILEGE_BIT ) != 0U )
        {
            xRunPrivileged = pdTRUE;
        }
        else
        {
            xRunPrivileged = pdFALSE;
        }
        uxPriority &= ~portPRIVILEGE_BIT; // 清除掉最高位
    #endif /* portUSING_MPU_WRAPPERS == 1 */

    // 如果tskSET_NEW_STACKS_TO_KNOWN_VALUE设置为1
    // 则可以将任务的栈填满一个固定的数，0xA5（因为0xa5二进制是1010101...交替的）
    // 这样就可以在调试的时候方便观察栈到底用了多少
    #if ( tskSET_NEW_STACKS_TO_KNOWN_VALUE == 1 )
    {
        /* Fill the stack with a known value to assist debugging. */
        ( void ) memset( pxNewTCB->pxStack, ( int ) tskSTACK_FILL_BYTE, ( size_t ) uxStackDepth * sizeof( StackType_t ) );
    }
    #endif /* tskSET_NEW_STACKS_TO_KNOWN_VALUE */

    // 栈的增长方向与对齐方式
    // portSTACK_GROWTH < 0表示栈向下增长
    #if ( portSTACK_GROWTH < 0 )
    {
        //注意：这里所谓的栈，只不过是我们声明的一个数组，被当作栈用罢了
        pxTopOfStack = &( pxNewTCB->pxStack[ uxStackDepth - ( configSTACK_DEPTH_TYPE ) 1 ] );
        pxTopOfStack = ( StackType_t * ) ( ( ( portPOINTER_SIZE_TYPE ) pxTopOfStack ) & ( ~( ( portPOINTER_SIZE_TYPE ) portBYTE_ALIGNMENT_MASK ) ) );

        //检查对齐方式是否正确（cortex-M为例的话，就是检查栈顶到栈底是否能按8字节对齐）
        configASSERT( ( ( ( portPOINTER_SIZE_TYPE ) pxTopOfStack & ( portPOINTER_SIZE_TYPE ) portBYTE_ALIGNMENT_MASK ) == 0U ) );
		//是否记录栈顶地址，可能会用于调试
        #if ( configRECORD_STACK_HIGH_ADDRESS == 1 )
        {
            pxNewTCB->pxEndOfStack = pxTopOfStack;
        }
        #endif /* configRECORD_STACK_HIGH_ADDRESS */
    }
    #else /* portSTACK_GROWTH */
    {
        pxTopOfStack = pxNewTCB->pxStack;
        pxTopOfStack = ( StackType_t * ) ( ( ( ( portPOINTER_SIZE_TYPE ) pxTopOfStack ) + portBYTE_ALIGNMENT_MASK ) & ( ~( ( portPOINTER_SIZE_TYPE ) portBYTE_ALIGNMENT_MASK ) ) );

        /* Check the alignment of the calculated top of stack is correct. */
        configASSERT( ( ( ( portPOINTER_SIZE_TYPE ) pxTopOfStack & ( portPOINTER_SIZE_TYPE ) portBYTE_ALIGNMENT_MASK ) == 0U ) );

        /* The other extreme of the stack space is required if stack checking is
         * performed. */
        pxNewTCB->pxEndOfStack = pxNewTCB->pxStack + ( uxStackDepth - ( configSTACK_DEPTH_TYPE ) 1 );
    }
    #endif /* portSTACK_GROWTH */

    /* Store the task name in the TCB. */
    if( pcName != NULL )
    {
        for( x = ( UBaseType_t ) 0; x < ( UBaseType_t ) configMAX_TASK_NAME_LEN; x++ )
        {
            pxNewTCB->pcTaskName[ x ] = pcName[ x ];

            // 注意不要无脑的使用memcpy按照configMAX_TASK_NAME_LEN拷贝任务名
            // 因为任务名的长度未必就是configMAX_TASK_NAME_LEN 
            // 虽然要保证任务名字符串长度是configMAX_TASK_NAME_LEN
            // 但是盲目拷贝，可能将不属于字符串的内容拷贝进来
            if( pcName[ x ] == ( char ) 0x00 )
            {
                break;
            }
            else
            {
                mtCOVERAGE_TEST_MARKER();
            }
        }
        pxNewTCB->pcTaskName[ configMAX_TASK_NAME_LEN - 1U ] = '\0';
    }
    else
    {
        // 钩子，可以用来输出一些debug信息，本质宏定义
        mtCOVERAGE_TEST_MARKER();
    }

    /* This is used as an array index so must ensure it's not too large. */
    configASSERT( uxPriority < configMAX_PRIORITIES );

    //确保优先级不会超出最大优先级
    if( uxPriority >= ( UBaseType_t ) configMAX_PRIORITIES )
    {
        uxPriority = ( UBaseType_t ) configMAX_PRIORITIES - ( UBaseType_t ) 1U;
    }
    else
    {
        // 钩子
        mtCOVERAGE_TEST_MARKER();
    }

    pxNewTCB->uxPriority = uxPriority;
    
    // 给任务的优先级做一个快照
    // 如果日后在抢占互斥量过程中，如果一直没抢到，那么这个任务的优先级会被临时提高
    // 防止它永远都没有抢到
    // 一旦这个过程结束，任务的优先级还是需要降低回去的
    #if ( configUSE_MUTEXES == 1 )
    {
        pxNewTCB->uxBasePriority = uxPriority;
    }
    #endif /* configUSE_MUTEXES */

    vListInitialiseItem( &( pxNewTCB->xStateListItem ) );
    vListInitialiseItem( &( pxNewTCB->xEventListItem ) );

    // 这是列表列的任务能还原成TCB的关键，让链表内的item指向本任务的TCB
    listSET_LIST_ITEM_OWNER( &( pxNewTCB->xStateListItem ), pxNewTCB );

    // 这段和事件有关，决定多个任务在等待互斥量的时候，谁先拿到
    listSET_LIST_ITEM_VALUE( &( pxNewTCB->xEventListItem ), ( TickType_t ) configMAX_PRIORITIES - ( TickType_t ) uxPriority );
    listSET_LIST_ITEM_OWNER( &( pxNewTCB->xEventListItem ), pxNewTCB );

    // 如果开了 MPU，把调用者传入的 MPU 区域参数 + 任务栈的地址范围，存到 TCB 里的 xMPUSettings 结构中。没开 MPU 就什么都不做，顺手消掉一个"未使用参数"的编译器警告。
    #if ( portUSING_MPU_WRAPPERS == 1 )
    {
        vPortStoreTaskMPUSettings( &( pxNewTCB->xMPUSettings ), xRegions, pxNewTCB->pxStack, uxStackDepth );
    }
    #else
    {
        /* Avoid compiler warning about unreferenced parameter. */
        ( void ) xRegions;
    }
    #endif

    // tls的支持，tls可以允许线程拥有自己的”线程变量“
    #if ( configUSE_C_RUNTIME_TLS_SUPPORT == 1 )
    {
        // 如果开了tls，就给任务一块线程局部内存xTLSBlock
        // 注意，具体怎么给，要用户自己实现，这里传入pxTopOfStack只是为了防止你要用这个参数
        configINIT_TLS_BLOCK( pxNewTCB->xTLSBlock, pxTopOfStack );
    }
    #endif
/*******************************************************************
 *在任务栈上伪造一套数据（或者说初始化一套数据）
 *让任务看起来就像是从运行状态被打断的一样，方便调度器调度
 *******************************************************************/
    #if ( portUSING_MPU_WRAPPERS == 1 )
    {
        /* 如果配置开启了栈溢出检查，走这里 */
        #if ( portHAS_STACK_OVERFLOW_CHECKING == 1 )
        {
            #if ( portSTACK_GROWTH < 0 ) // 栈增长方向向下
            {
                // 如果开启了MPU，则初始化栈的时候，需要指明该任务是否以特权模式运行
                // 同时需要初始化MPU设置
                pxNewTCB->pxTopOfStack = pxPortInitialiseStack( pxTopOfStack, pxNewTCB->pxStack, pxTaskCode, pvParameters, xRunPrivileged, &( pxNewTCB->xMPUSettings ) );
            }
            #else /* portSTACK_GROWTH */ // 栈增长方向向上
            {
                pxNewTCB->pxTopOfStack = pxPortInitialiseStack( pxTopOfStack, pxNewTCB->pxEndOfStack, pxTaskCode, pvParameters, xRunPrivileged, &( pxNewTCB->xMPUSettings ) );
            }
            #endif /* portSTACK_GROWTH */
        }
        #else /* portHAS_STACK_OVERFLOW_CHECKING */
        {
            /* 如果没有自动的栈溢出检查，则需要手动计算栈顶，放在TCB中 */
            /* 这个版本也是之前学习野火课程的时候所使用的 */
            pxNewTCB->pxTopOfStack = pxPortInitialiseStack( pxTopOfStack, pxTaskCode, pvParameters, xRunPrivileged, &( pxNewTCB->xMPUSettings ) );
        }
        #endif /* portHAS_STACK_OVERFLOW_CHECKING */
    }
    #else /* portUSING_MPU_WRAPPERS */ // 在没有配置MPU的情况下，进行栈配置，将栈伪装成被打断执行后的样子
    {
        /* If the port has capability to detect stack overflow,
         * pass the stack end address to the stack initialization
         * function as well. */
        #if ( portHAS_STACK_OVERFLOW_CHECKING == 1 )
        {
            #if ( portSTACK_GROWTH < 0 )
            {
                pxNewTCB->pxTopOfStack = pxPortInitialiseStack( pxTopOfStack, pxNewTCB->pxStack, pxTaskCode, pvParameters );
            }
            #else /* portSTACK_GROWTH */
            {
                pxNewTCB->pxTopOfStack = pxPortInitialiseStack( pxTopOfStack, pxNewTCB->pxEndOfStack, pxTaskCode, pvParameters );
            }
            #endif /* portSTACK_GROWTH */
        }
        #else /* portHAS_STACK_OVERFLOW_CHECKING */
        {
            pxNewTCB->pxTopOfStack = pxPortInitialiseStack( pxTopOfStack, pxTaskCode, pvParameters );
        }
        #endif /* portHAS_STACK_OVERFLOW_CHECKING */
    }
    #endif /* portUSING_MPU_WRAPPERS */

    /* Initialize task state and task attributes. */
    #if ( configNUMBER_OF_CORES > 1 )
    {
        pxNewTCB->xTaskRunState = taskTASK_NOT_RUNNING;

        /* Is this an idle task? */
        if( ( ( TaskFunction_t ) pxTaskCode == ( TaskFunction_t ) prvIdleTask ) || ( ( TaskFunction_t ) pxTaskCode == ( TaskFunction_t ) prvPassiveIdleTask ) )
        {
            pxNewTCB->uxTaskAttributes |= taskATTRIBUTE_IS_IDLE;
        }
    }
    #endif /* #if ( configNUMBER_OF_CORES > 1 ) */

    if( pxCreatedTask != NULL )
    {
        /* Pass the handle out in an anonymous way.  The handle can be used to
         * change the created task's priority, delete the created task, etc.*/
        *pxCreatedTask = ( TaskHandle_t ) pxNewTCB;
    }
    else
    {
        mtCOVERAGE_TEST_MARKER();
    }
}
```

#### vTaskStartScheduler

**负责启动调度器**



```c
void vTaskStartScheduler( void )
{
    BaseType_t xReturn;

    traceENTER_vTaskStartScheduler(); // 钩子

    // 如果cpu是多核的并且开启了核亲和
    #if ( configUSE_CORE_AFFINITY == 1 ) && ( configNUMBER_OF_CORES > 1 )
    {
        // 确保亲和性能表示的数量，大于等于，实际核心数量
        configASSERT( ( sizeof( UBaseType_t ) * taskBITS_PER_BYTE ) >= configNUMBER_OF_CORES );
    }
    #endif /* #if ( configUSE_CORE_AFFINITY == 1 ) && ( configNUMBER_OF_CORES > 1 ) */

    /*********************************
     * 创建空闲任务
     *********************************/
    xReturn = prvCreateIdleTasks();// 创建空闲任务

    /**********************************
     * 视情况初始化软件定时子系统
     * 这个子系统只有一个作用：创建定时服务任务
     *********************************/
    #if ( configUSE_TIMERS == 1 )
    {
        if( xReturn == pdPASS ) // 表示之前的prvCreateIdleTasks执行成功
        {
            xReturn = xTimerCreateTimerTask(); // 创建定时器任务
        }
        else
        {
            mtCOVERAGE_TEST_MARKER(); // 钩子
        }
    }
    #endif /* configUSE_TIMERS */

    if( xReturn == pdPASS )
    {
        // 如果配置了FREERTOS_TASKS_C_ADDITIONS_INIT就会执行freertos_tasks_c_additions_init
        // 这个函数是给用户实现的，可以用来做一些用户定制的工作
        #ifdef FREERTOS_TASKS_C_ADDITIONS_INIT
        {
            freertos_tasks_c_additions_init(); // 用户可以通过实现这个函数实现一些自定义功能
        }
        #endif
        
        // 这里通过一个宏关闭中断，防止在调用xPortStartScheduler之前由于tick中断
        // 或其他中断导致程序异常，实际的宏实现如下：
        // #define portDISABLE_INTERRUPTS()    __asm volatile ( " cpsid i " ::: "memory" )
        portDISABLE_INTERRUPTS();

        // TLS是运行过程中，每个线程都可以拥有属于自己独立的全局变量
        #if ( configUSE_C_RUNTIME_TLS_SUPPORT == 1 )
        {
            // 该函数的主要作用是，C运行时的TLS环境从启动状态切换到任务状态
            // 将第一个被执行的任务的环境赋给TLS
            configSET_TLS_BLOCK( pxCurrentTCB->xTLSBlock );
        }
        #endif

        xNextTaskUnblockTime = portMAX_DELAY; // 将下一个非阻塞任务该醒来的时间设置为
        								   // 最大延时时间
        								   // 因为启动的时候，肯定是没有延时任务的
        xSchedulerRunning = pdTRUE; // 调度器运行指示
        xTickCount = ( TickType_t ) configINITIAL_TICK_COUNT; // tick计数器

        // 如果需要 运行时间统计 功能， 必须有一个高精度的硬件定时器，并且通过
        // 下面这个函数初始化这个硬件
        // 如果开启了宏：FreeRTOSConfig.h/configGENERATE_RUN_TIME_STATS 为 1
        // portCONFIGURE_TIMER_FOR_RUN_TIME_STATS()是一个移植层宏定义，是需要用户实现的
        // 需要在这个宏里，初始化高精度定时器，并且启动这个定时器
```

#### prvCreateIdleTasks

**该函数负责创建空闲任务**

### 3. 启动任务调度

#### xPortStartScheduler

**这个函数是真正启动调度器的函数**

在开始分析代码之前，需要了解一些前提；



##### MPU机制

MPU是一个硬件机制，该机制主要有2个作用：

1. MPU可以将特定位置的内存设置为：非特权状态，不可赋值
2. 如果某个非特权指针，访问到了这块内存，并且尝试给该区域赋值，会触发中断

MPU 类型寄存器 TYPER                    0xE000ED90
MPU 控制寄存器 CTRL                       0xE000ED94
MPU region 号寄存器 RNR                 0xE000ED98
MPU region 基地址寄存器 RBAR       0xE000ED9C
MPU region 属性寄存器 RASR           0xE000EDA0

###### TYPER 0xE000ED90

TYPER 表示这块SoC的MPU支持多少data region



###### MPU_CTRL 0xE000ED94

| 位数 |    名字    | 类型 |               作用               |
| :--: | :--------: | :--: | :------------------------------: |
|  2   | PRIVDEFENA | 读写 | 是否为特权级打开 缺省存储区 映射 |
|  1   |  HFNMIENA  | 读写 |   在hardfault中是否默认使能MPU   |
|  0   |   ENABLE   | 读写 |             使能MPU              |

PRIDEFENA操作的，没有被定义为region的区域，也就是“背景”，如果PRIVDEFENA = 1，那么背景就只允许特权级别访问，如果PRIVDEFENA = 0，那么背景就禁止访问；



##### 硬件中断与RTOS中断

这个函数里涉及到了一个问题：RTOS的中断 和 硬件的中断

以Cortex-M为例，硬件支持8位NVIC，也就是最高255级中断优先级，其中数字越小，中断优先级越高；

RTOS又有自己的一套优先级，这套优先级的级数由软件设定，configMAX_PRIORITIES原则上是没有什么限制的（大到溢出的数不行），且对RTOS而言优先级数字越小，优先级越低；

对于RTOS而言，他自己的那一套优先级实际是逻辑上的优先级，如果硬件中断到了，那么SoC肯定优先处理硬件中断（进入ISR--中断服务程序），RTOS是管不了的。这就引出了一个问题：如果RTOS在执行一个任务，此时来了一个硬件中断，ISR中修改了RTOS任务的数据内存，这样RTOS在不知情的情况下，内存被踩了；

为了解决这个问题，RTOS使用了一个硬件机制：屏蔽中断

Cortex为例，这个cpu提供了BASEPRI寄存器，通过设置这个寄存器的值，确立一个**界限**，可以屏蔽一部分优先级较低的硬件中断；

硬件中断优先级高于这个界限的中断，不应该直接修改RTOS内核的数据，只做硬件操作就好，如果要写内存，写他自己的buffer，不能调用FromISR API（这里的不能实际上是不应调用FromISR API，而非无法调用FromISR API，硬要调用的话，没人拦得住）

硬件中断优先级低于这个界限的中断，可以调用 FromISR API；

FromISR API是一组API而非一个API，这组API负责队列、任务通知、信号量、事件组...的操作，这保证了，我们在**需要保证原子操作的情况下，可以通过屏蔽一部分可能同样会操作内核对象的中断，保证内核对象的原子操作；**



##### 抢占优先级 和 子优先级

！！注，这是一个硬件功能

如果NVIC是8位，cortex会把这8位分成两部分，一部分是 **抢占优先级** 另一部分是**子优先级**；并不是永远都有

比如：高7位是抢占优先级，低1位是子优先级

没有子优先级的情况下：

进程A 抢占优先级是2(0010)

进程B 抢占优先级是3(0011)

进程C 抢占优先级是3(0011)

当进程B运行时，进程A可以随意打断进程B；尽管C和B的优先级都是3，但是B运行时，C不能打断B；C运行时也是同理；

在有子优先级的情况下：

进程A 抢占优先级是2 (0010X)

进程B 抢占优先级是3 子优先级0(0011 0)

进程C 抢占优先级是3 子优先级1(0011 1)

此时B运行时，A仍旧可以随意打断；与之前不同的是，现在C也打断运行中的B；



##### 子优先级（PRIGROUP）会造成问题

Cortex是根据 BASEPRI 寄存器 的纸面数字进行中断屏蔽的

一旦子优先级生效，会出现：

BASEPRI 设置为4时，优先级5和优先级4的抢占优先级一样，子优先级不同，但是其中一个会被屏蔽，而另一个却不会被屏蔽；

为了解决这个问题，FreeRTOS会在NIVC==8（也就是子优先级生效）的情况下，屏蔽掉最低的一位；



##### 代码分析

```c

BaseType_t xPortStartScheduler( void ) /* PRIVILEGED_FUNCTION */
{
    // 调度是通过SVC中断与PendSV中断实现的
    // 而SVC和PendSV中断服务函数，放在中断向量表里，因此这里衍生出两种不同的调度方法
    // 本质上是，有两种安排中断服务函数的方式：
    // 方式1：直接路由--也就是直接在异常向量表里实现，这种方法的svc和pendsv服务
    //        一般已经由FreeRTOS帮你写好了，不用管（对于常见的SoC来说）
    // 方式2：间接路由--这种方式需要修改异常向量表，自己写一个wrapper函数
    //        把自己写的wrapper函数放到异常向量表里，然后在wrapper种调用
    //        FreeRTOS的SVC_Handler或者PendSV_Handler；这种方式的优势是
    //        你可以在执行真正负责任务调度的函数之前，插入一段你的定制代码
    // 注意：如果使用间接路由，需要将configCHECK_HANDLER_INSTALLATION设置为0
    //      否则程序会执行接下来检查，导致执行失败
    #if ( configCHECK_HANDLER_INSTALLATION == 1 ) // 直接路由
    {
        const portISR_t * const pxVectorTable = portSCB_VTOR_REG;

        // 通过断言判断SVC的中断处理函数和PendSV的中断处理函数安装是否正确
        // 这里不检查SysTick的中断处理函数，是因为SysTick有可能是由用户自己实现的
        // 如果SDK的异常向量表是可以自己修改的，那么改了异常向量表的偏移地址，也能会
        // 引发断言的失败
        configASSERT( pxVectorTable[ portVECTOR_INDEX_SVC ] == SVC_Handler );
        configASSERT( pxVectorTable[ portVECTOR_INDEX_PENDSV ] == PendSV_Handler );
    }
    #endif /* configCHECK_HANDLER_INSTALLATION */

    // 接下来的内容，是判断 硬件异常等级（注意，硬件异常等级！=RTOS的软件异常等级）
    // 不同的SoC的NVIC支持的异常等级不同，有的NIVC是8位，有的是4位
    // ！！！！ 请注意：接下来的代码块有一个条件portHAS_ARMV8M_MAIN_EXTENSION == 1
    //                只有ARMV8配置的硬件才会执行下述代码
    #if ( ( configASSERT_DEFINED == 1 ) && ( portHAS_ARMV8M_MAIN_EXTENSION == 1 ) )
    {
        volatile uint32_t ulImplementedPrioBits = 0;
        volatile uint8_t ucMaxPriorityValue;

        // 下两行代码，通过向portNVIC_SHPR2_REG（系统handler优先级寄存器）的高8位写1
        // 假如写进去的0xff，但是读出来的是0x0f，那么就说明，这个SoC就只支持4位优先级(也就是0~15)
        // 那么ucMaxPriorityValue 就等于0x0F 代表最大的优先级级数是15；
        // 这个写法是基于ARM NIVC的，如果你的项目中，硬件不是这样的，那么要根据实际的硬件写代码；
        portNVIC_SHPR2_REG = 0xFF000000; // 向寄存器写内容
        ucMaxPriorityValue = ( uint8_t ) ( ( portNVIC_SHPR2_REG & 0xFF000000 ) >> 24 );//回读

        /******************************************************************
         * 注意：configMAX_SYSCALL_INTERRUPT_PRIORITY是一个很重要的配置
         *      这个宏 划定一个一个 “安全界限” 
         ******************************************************************/
        // 下面的一行代码，是为了将用户配置的configMAX_SYSCALL_INTERRUPT_PRIORITY与实际硬件支持的
        // 优先级对齐；如果不进行对齐，可能会发生如下情况：
        // 举例：假设用户设置的configMAX_SYSCALL_INTERRUPT_PRIORITY是0xF0
        // 而实际上的硬件支持的优先级只有4位，也就是最大0x0F，此时将0xF0写入寄存器，会发生溢出
        // 最终写入寄存器的是0，实际写入的和预料中写入的产生了错误；
        ucMaxSysCallPriority = configMAX_SYSCALL_INTERRUPT_PRIORITY & ucMaxPriorityValue;

        // 断言：ucMaxSysCallPriority不是0
        // 如果经过上一步对齐之后，ucMaxSysCallPriority变成了0，就会拦截所有硬中断，导致一些高优先级的
        // 硬中断也无法被触发，影响实时性；
        // 这里可以拦截两种情况：
        // 1，configMAX_SYSCALL_INTERRUPT_PRIORITY本身就是0
        // 2. 经过计算后的ucMaxSysCallPriority是0
        // 这两种情况都属于configMAX_SYSCALL_INTERRUPT_PRIORITY设置不当
        configASSERT( ucMaxSysCallPriority );

        /* Check that the bits not implemented in hardware are zero in
         * configMAX_SYSCALL_INTERRUPT_PRIORITY. */
        // 这句判断 configMAX_SYSCALL_INTERRUPT_PRIORITY 是否能够正确的作用于NVIC上
        // 假设 ucMaxPriorityValue = 0x0F，经过uint32扩展 和 取反之后，变成了0xFFFF FFF0
        // 如果 configMAX_SYSCALL_INTERRUPT_PRIORITY 的值大于0x0F，比如其值为0x1F
        // 此时位与之后，结果就不是0，说明configMAX_SYSCALL_INTERRUPT_PRIORITY中
        // 存在NVIC不支持的位；
        configASSERT( ( configMAX_SYSCALL_INTERRUPT_PRIORITY & 
                       ( uint8_t ) ( ~( uint32_t ) ucMaxPriorityValue ) ) == 0U );

        /* Calculate the maximum acceptable priority group value for the number
         * of bits read back. */
        // 注： portTOP_BIT_OF_BYTE在ARMV8的配置下，是0x80（1000 0000）
        // 下面的循环代码，是在判断，ucMaxPriorityValue需要左移多少次才能变成 1000 0000
        // 假设 ucMaxPriorityValue = 0x0F， 那么需要左移8次才能让 ucMaxPriorityValue = 0x80
        // ulImplementedPrioBits 也就等于 8
        // 当然 如果ucMaxPriorityValue = 0xFF，也是需要左移8次才能让 ucMaxPriorityValue = 0x80
        // 真正会让 ulImplementedPrioBits 变化的是 0XFE这类 最低位不是1的值；
        while( ( ucMaxPriorityValue & portTOP_BIT_OF_BYTE ) == portTOP_BIT_OF_BYTE )
        {
            ulImplementedPrioBits++;
            ucMaxPriorityValue <<= ( uint8_t ) 0x01;
        }

        if( ulImplementedPrioBits == 8 )
        {
            // 1. 如果NVIC是完整的8位，那么其最低为一定是子优先级位
            // 2. 在1所述的情况下实际上NVIC中断优先级5和NVIC中断优先级4，在抢占优先级上是相同的
            //    如果将 “中断的安全界限” 设为类似于5这样的数
            //    就会引起歧义，因此这里要判断一下 “安全界限” 的子优先级是否是0
            //    如果不是0就卡住程序不再向下运行
            configASSERT( ( configMAX_SYSCALL_INTERRUPT_PRIORITY & 0x1U ) == 0U );
            ulMaxPRIGROUPValue = 0; // 意味着子优先级只允许为0
        }
        else
        {
            // 如果不是8位NVIC，那么需要手动计算一下 子优先级 占多少位
            // 以 ucMaxPriorityValue = 0xF0为例（低4位=0）
            // 此时ulImplementedPrioBits = 3
            // portMAX_PRIGROUP_BITS - ulImplementedPrioBits的结果为 4
            // 这代表着 写这款ARMV8 的 MCU的时候，低4位的值都应当保持0;
            // 总结： 计算ulMaxPRIGROUPValue是为了让configMAX_SYSCALL_INTERRUPT_PRIORITY
            //        这个值保持正确的写进NVIC
            ulMaxPRIGROUPValue = portMAX_PRIGROUP_BITS - ulImplementedPrioBits;
        }

        /* Shift the priority group value back to its position within the AIRCR
         * register. */
        // 以 NVIC低4位需要全=0为例
        // 4 << 8  即 0000 0000 0000 0100 << 8 = 0000 0100 0000 0000
        // 0000 0100 0000 0000 & 0000 0111 0000 0000 = 0000 0111 0000 0000
        ulMaxPRIGROUPValue <<= portPRIGROUP_SHIFT; // portPRIGROUP_SHIFT 8UL
        ulMaxPRIGROUPValue &= portPRIORITY_GROUP_MASK; // portPRIORITY_GROUP_MASK (0x07UL<<8UL)
    }
    #endif /* #if ( ( configASSERT_DEFINED == 1 ) && ( portHAS_ARMV8M_MAIN_EXTENSION == 1 ) ) */

    // 接下来是寄存器操作
    // 将PendSV和SysTick的优先级降到最低
    // 将SVC的优先级提到最高
    portNVIC_SHPR3_REG |= portNVIC_PENDSV_PRI;
    portNVIC_SHPR3_REG |= portNVIC_SYSTICK_PRI;
    portNVIC_SHPR2_REG = 0;

    #if ( configENABLE_MPU == 1 )
    {
        // 设置内存保护单元 暂时不做展开学习
        prvSetupMPU();
    }
    #endif /* configENABLE_MPU */

    // 启动定时器
    vPortSetupTimerInterrupt();

    /* Initialize the critical nesting count ready for the first task. */
    ulCriticalNesting = 0;

    #if ( ( configENABLE_MPU == 1 ) && ( configUSE_MPU_WRAPPERS_V1 == 0 ) )
    {
        xSchedulerRunning = pdTRUE;
    }
    #endif /* ( ( configENABLE_MPU == 1 ) && ( configUSE_MPU_WRAPPERS_V1 == 0 ) ) */

    /* Start the first task. */
    vStartFirstTask();

    /* Should never get here as the tasks will now be executing. Call the task
     * exit error function to prevent compiler warnings about a static function
     * not being called in the case that the application writer overrides this
     * functionality by defining configTASK_RETURN_ADDRESS. Call
     * vTaskSwitchContext() so link time optimization does not remove the
     * symbol. */
    vTaskSwitchContext();
    prvTaskExitError();

    /* Should not get here. */
    return 0;
}
```

###### Trust zone机制

接下来的代码里会出现Trust zone机制

这个机制是ARM提供的硬件机制，将SoC的所有资源（CPU 外设 内存 中断 e.t.c）划分为安全世界和普通世界，安全世界运行可信代码（如安全启动、密钥管理、DRM、支付、生物识别比对）；普通世界则运行常规操作系统（如 Linux、Android、FreeRTOS）和应用程序

- 处理器通过 **`SCR`（Secure Configuration Register）** 或 **`NS`（Non-Secure）位** 标识当前世界。
- 安全世界可以随时访问非安全世界的资源，但**非安全世界永远无法直接访问安全世界的资源**（除非安全世界显式导出）。
- 两个世界之间的切换通过 **`SMC`（Secure Monitor Call）指令** 或 **异常/中断** 触发。



###### vPortSVCHandler_C

这个函数是有参数的，根据函数调用规范，这个参数应当放在r0寄存器中，而在SVC_Handler这个汇编函数中，r0确实已经被赋值为了栈地址；

该函数会处理 SVC中断编号：100、101、102、103这四种情况

这里暂时先只分析编号102——portSVC_START_SCHEDULER，其中**vRestoreContextOfFirstTask**是重点

```c
void vPortSVCHandler_C( uint32_t * pulCallerStackAddress ) /* PRIVILEGED_FUNCTION portDONT_DISCARD */
{
    #if ( ( configENABLE_MPU == 1 ) && ( configUSE_MPU_WRAPPERS_V1 == 1 ) )
        #if defined( __ARMCC_VERSION ) // 判断用的是不是ARM编译器
            // 如果使用的是ARMCC，则使指针的形式声明如下两个变量
            extern uint32_t * __syscalls_flash_start__;
            extern uint32_t * __syscalls_flash_end__;
        #else
            // 如果使用的是其他C编译器，则以空数组的形式声明变量
            extern uint32_t __syscalls_flash_start__[];
            extern uint32_t __syscalls_flash_end__[];
        #endif /* defined( __ARMCC_VERSION ) */
    #endif /* ( configENABLE_MPU == 1 ) && ( configUSE_MPU_WRAPPERS_V1 == 1 ) */

    uint32_t ulPC;

    #if ( configENABLE_TRUSTZONE == 1 )
        uint32_t ulR0, ulR1;
        extern TaskHandle_t pxCurrentTCB;
        #if ( configENABLE_MPU == 1 )
            uint32_t ulControl, ulIsTaskPrivileged;
        #endif /* configENABLE_MPU */
    #endif /* configENABLE_TRUSTZONE */
    uint8_t ucSVCNumber;

    /* Register are stored on the stack in the following order - R0, R1, R2, R3,
     * R12, LR, PC, xPSR. */
    ulPC = pulCallerStackAddress[ portOFFSET_TO_PC ];
    ucSVCNumber = ( ( uint8_t * ) ulPC )[ -2 ];

    switch( ucSVCNumber )
    {
        #if ( configENABLE_TRUSTZONE == 1 )
            case portSVC_ALLOCATE_SECURE_CONTEXT: // 100

                /* R0 contains the stack size passed as parameter to the
                 * vPortAllocateSecureContext function. */
                ulR0 = pulCallerStackAddress[ 0 ];

                #if ( configENABLE_MPU == 1 )
                {
                    /* Read the CONTROL register value. */
                    __asm volatile ( "mrs %0, control"  : "=r" ( ulControl ) );

                    /* The task that raised the SVC is privileged if Bit[0]
                     * in the CONTROL register is 0. */
                    ulIsTaskPrivileged = ( ( ulControl & portCONTROL_PRIVILEGED_MASK ) == 0 );

                    /* Allocate and load a context for the secure task. */
                    xSecureContext = SecureContext_AllocateContext( ulR0, ulIsTaskPrivileged, pxCurrentTCB );
                }
                #else /* if ( configENABLE_MPU == 1 ) */
                {
                    /* Allocate and load a context for the secure task. */
                    xSecureContext = SecureContext_AllocateContext( ulR0, pxCurrentTCB );
                }
                #endif /* configENABLE_MPU */

                configASSERT( xSecureContext != securecontextINVALID_CONTEXT_ID );
                SecureContext_LoadContext( xSecureContext, pxCurrentTCB );
                break;

            case portSVC_FREE_SECURE_CONTEXT: // 101

                /* R0 contains TCB being freed and R1 contains the secure
                 * context handle to be freed. */
                ulR0 = pulCallerStackAddress[ 0 ];
                ulR1 = pulCallerStackAddress[ 1 ];

                /* Free the secure context. */
                SecureContext_FreeContext( ( SecureContextHandle_t ) ulR1, ( void * ) ulR0 );
                break;
        #endif /* configENABLE_TRUSTZONE */
            /*-----------------------------------------------------------------------*/
            case portSVC_START_SCHEDULER: // 102（注意！！！！这个是默认情况）
            #if ( configENABLE_TRUSTZONE == 1 )
            {
                /* De-prioritize the non-secure exceptions so that the
                 * non-secure pendSV runs at the lowest priority. */
                SecureInit_DePrioritizeNSExceptions();

                /* Initialize the secure context management system. */
                SecureContext_Init();
            }
            #endif /* configENABLE_TRUSTZONE */

            #if ( configENABLE_FPU == 1 )
            {
                /* Setup the Floating Point Unit (FPU). */
                prvSetupFPU();
            }
            #endif /* configENABLE_FPU */

            /* Setup the context of the first task so that the first task starts
             * executing. */
            vRestoreContextOfFirstTask(); //!!!!!!!!!这个函数是重点
            break;
            /*----------------------------------------------------------------------*/
            #if ( ( configENABLE_MPU == 1 ) && ( configUSE_MPU_WRAPPERS_V1 == 1 ) )
                case portSVC_RAISE_PRIVILEGE: // 103

                    /* Only raise the privilege, if the svc was raised from any of
                     * the system calls. */
                    if( ( ulPC >= ( uint32_t ) __syscalls_flash_start__ ) &&
                        ( ulPC <= ( uint32_t ) __syscalls_flash_end__ ) )
                    {
                        vRaisePrivilege();
                    }
                    break;
            #endif /* ( configENABLE_MPU == 1 ) && ( configUSE_MPU_WRAPPERS_V1 == 1 ) */

            #if ( configENABLE_MPU == 1 )
                case portSVC_YIELD:
                    vPortYield();
                    break;
            #endif /* configENABLE_MPU == 1 */

        default:
            /* Incorrect SVC call. */
            configASSERT( pdFALSE );
    }
}
```

```assembly
        ldmia r2!, {r0, r3-r6}   # 把r0、r3到r6从当前任务的栈里加载到对应寄存器里
        subs r2, #20             # 由于上一行使用了回写（！），栈帧再往后减20个字节，回到原点
        # 接下来的几句话都是在将栈里的内容写入寄存器
        msr psp, r3              # 把r3的值(original PSP)写入psp寄存器中
        msr control, r5          
        mov lr, r6
        ldr r4, =xSecureContext             /* Read the location of xSecureContext i.e. &( xSecureContext ). */
        str r0, [r4]                        /* Restore xSecureContext. */

    restore_general_regs_first_task:
        subs r2, #32
        ldmia r2!, {r4-r7}                  /* r4-r7 contain half of the hardware saved context. */
        stmia r3!, {r4-r7}                  /* Copy half of the the hardware saved context on the task stack. */
        ldmia r2!, {r4-r7}                  /* r4-r7 contain rest half of the hardware saved context. */
        stmia r3!, {r4-r7}               /* Copy rest half of the the hardware saved context on the task stack. */
        subs r2, #48
        ldmia r2!, {r4-r7}                  /* Restore r8-r11. */
        mov r8, r4                          /* r8 = r4. */
        mov r9, r5                          /* r9 = r5. */
        mov r10, r6                         /* r10 = r6. */
        mov r11, r7                         /* r11 = r7. */
        subs r2, #32
        ldmia r2!, {r4-r7}                  /* Restore r4-r7. */
        subs r2, #16

    restore_context_done_first_task:
       str r2, [r1]     /* Save the location where the context should be saved next as the first member of TCB. */
       bx lr

/*----------------------接下来的------------------------*/
#else /* configENABLE_MPU */

vRestoreContextOfFirstTask:
    ldr  r2, =pxCurrentTCB                  /* Read the location of pxCurrentTCB i.e. &( pxCurrentTCB ). */
    ldr  r3, [r2]                           /* Read pxCurrentTCB. */
    ldr  r0, [r3]                           /* Read top of stack from TCB - The first item in pxCurrentTCB is the task top of stack. */

    ldm  r0!, {r1-r3}                       /* Read from stack - r1 = xSecureContext, r2 = PSPLIM and r3 = EXC_RETURN. */
    ldr  r4, =xSecureContext
    str  r1, [r4]                           /* Set xSecureContext to this task's value for the same. */
    movs r1, #2                             /* r1 = 2. */
    msr  CONTROL, r1                        /* Switch to use PSP in the thread mode. */
    adds r0, #32                            /* Discard everything up to r0. */
    msr  psp, r0                            /* This is now the new top of stack to use in the task. */
    isb
    bx   r3                                 /* Finally, branch to EXC_RETURN. */

#endif /* configENABLE_MPU */
```

````c
// 通过中断服务调用这个函数，
void vSystemCallEnter( uint32_t * pulTaskStack,
                           uint32_t ulLR,
                           uint8_t ucSystemCallNumber ) /* PRIVILEGED_FUNCTION */
    {
        extern TaskHandle_t pxCurrentTCB;
        extern UBaseType_t uxSystemCallImplementations[ NUM_SYSTEM_CALLS ];
        xMPU_SETTINGS * pxMpuSettings;
        uint32_t * pulSystemCallStack;
        uint32_t ulStackFrameSize, ulSystemCallLocation, i;

        #if defined( __ARMCC_VERSION )
            /* Declaration when these variable are defined in code instead of being
             * exported from linker scripts. */
            extern uint32_t * __syscalls_flash_start__;
            extern uint32_t * __syscalls_flash_end__;
        #else
            /* Declaration when these variable are exported from linker scripts. */
            extern uint32_t __syscalls_flash_start__[];
            extern uint32_t __syscalls_flash_end__[];
        #endif /* #if defined( __ARMCC_VERSION ) */

        ulSystemCallLocation = pulTaskStack[ portOFFSET_TO_PC ];
        pxMpuSettings = xTaskGetMPUSettings( pxCurrentTCB );

        /* Checks:
         * 1. SVC is raised from the system call section (i.e. application is
         *    not raising SVC directly).
         * 2. pxMpuSettings->xSystemCallStackInfo.pulTaskStack must be NULL as
         *    it is non-NULL only during the execution of a system call (i.e.
         *    between system call enter and exit).
         * 3. System call is not for a kernel API disabled by the configuration
         *    in FreeRTOSConfig.h.
         * 4. We do not need to check that ucSystemCallNumber is within range
         *    because the assembly SVC handler checks that before calling
         *    this function.
         */
        if( ( ulSystemCallLocation >= ( uint32_t ) __syscalls_flash_start__ ) &&
            ( ulSystemCallLocation <= ( uint32_t ) __syscalls_flash_end__ ) &&
            ( pxMpuSettings->xSystemCallStackInfo.pulTaskStack == NULL ) &&
            ( uxSystemCallImplementations[ ucSystemCallNumber ] != ( UBaseType_t ) 0 ) )
        {
            pulSystemCallStack = pxMpuSettings->xSystemCallStackInfo.pulSystemCallStack;

            #if ( ( configENABLE_FPU == 1 ) || ( configENABLE_MVE == 1 ) )
            {
                if( ( ulLR & portEXC_RETURN_STACK_FRAME_TYPE_MASK ) == 0UL )
                {
                    /* Extended frame i.e. FPU in use. */
                    ulStackFrameSize = 26;
                    __asm volatile (
                        " vpush {s0}         \n" /* Trigger lazy stacking. */
                        " vpop  {s0}         \n" /* Nullify the affect of the above instruction. */
                        ::: "memory"
                        );
                }
                else
                {
                    /* Standard frame i.e. FPU not in use. */
                    ulStackFrameSize = 8;
                }
            }
            #else /* if ( ( configENABLE_FPU == 1 ) || ( configENABLE_MVE == 1 ) ) */
            {
                ulStackFrameSize = 8;
            }
            #endif /* configENABLE_FPU || configENABLE_MVE */

            /* Make space on the system call stack for the stack frame. */
            pulSystemCallStack = pulSystemCallStack - ulStackFrameSize;

            /* Copy the stack frame. */
            for( i = 0; i < ulStackFrameSize; i++ )
            {
                pulSystemCallStack[ i ] = pulTaskStack[ i ];
            }

            /* Store the value of the Link Register before the SVC was raised.
             * It contains the address of the caller of the System Call entry
             * point (i.e. the caller of the MPU_<API>). We need to restore it
             * when we exit from the system call. */
            pxMpuSettings->xSystemCallStackInfo.ulLinkRegisterAtSystemCallEntry = pulTaskStack[ portOFFSET_TO_LR ];
            /* Store the value of the PSPLIM register before the SVC was raised.
             * We need to restore it when we exit from the system call. */
            #if ( portUSE_PSPLIM_REGISTER == 1 )
            {
                __asm volatile ( "mrs %0, psplim" : "=r" ( pxMpuSettings->xSystemCallStackInfo.ulStackLimitRegisterAtSystemCallEntry ) );
            }
            #endif

            /* Use the pulSystemCallStack in thread mode. */
            __asm volatile ( "msr psp, %0" : : "r" ( pulSystemCallStack ) );
            #if ( portUSE_PSPLIM_REGISTER == 1 )
            {
                __asm volatile ( "msr psplim, %0" : : "r" ( pxMpuSettings->xSystemCallStackInfo.pulSystemCallStackLimit ) );
            }
            #endif

            /* Start executing the system call upon returning from this handler. */
            pulSystemCallStack[ portOFFSET_TO_PC ] = uxSystemCallImplementations[ ucSystemCallNumber ];
            /* Raise a request to exit from the system call upon finishing the
             * system call. */
            pulSystemCallStack[ portOFFSET_TO_LR ] = ( uint32_t ) vRequestSystemCallExit;

            /* Remember the location where we should copy the stack frame when we exit from
             * the system call. */
            pxMpuSettings->xSystemCallStackInfo.pulTaskStack = pulTaskStack + ulStackFrameSize;

            /* Record if the hardware used padding to force the stack pointer
             * to be double word aligned. */
            if( ( pulTaskStack[ portOFFSET_TO_PSR ] & portPSR_STACK_PADDING_MASK ) == portPSR_STACK_PADDING_MASK )
            {
                pxMpuSettings->ulTaskFlags |= portSTACK_FRAME_HAS_PADDING_FLAG;
            }
            else
            {
                pxMpuSettings->ulTaskFlags &= ( ~portSTACK_FRAME_HAS_PADDING_FLAG );
            }

            /* We ensure in pxPortInitialiseStack that the system call stack is
             * double word aligned and therefore, there is no need of padding.
             * Clear the bit[9] of stacked xPSR. */
            pulSystemCallStack[ portOFFSET_TO_PSR ] &= ( ~portPSR_STACK_PADDING_MASK );

            /* Raise the privilege for the duration of the system call. */
            __asm volatile (
                " mrs r0, control     \n" /* Obtain current control value. */
                " movs r1, #1         \n" /* r1 = 1. */
                " bics r0, r1         \n" /* Clear nPRIV bit. */
                " msr control, r0     \n" /* Write back new control value. */
                ::: "r0", "r1", "memory"
                );
        }
    }
````
